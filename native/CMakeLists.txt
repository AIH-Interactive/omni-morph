if(NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Native acceleration requires a 64-bit process")
endif()
find_package(JNI REQUIRED)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(CMAKE_POSITION_INDEPENDENT_CODE ON)

string(TOLOWER "${CMAKE_SYSTEM_PROCESSOR}" OMNIMORPH_PROCESSOR)
if(OMNIMORPH_PROCESSOR MATCHES "^(aarch64|arm64)$" AND APPLE)
    set(OMNIMORPH_ARCH OMNIMORPH_ARM64)
    set(OMNIMORPH_KERNELS src/neon.cpp)
elseif(OMNIMORPH_PROCESSOR MATCHES "^(amd64|x86_64)$")
    set(OMNIMORPH_ARCH OMNIMORPH_X64)
    set(OMNIMORPH_KERNELS src/sse2.cpp src/avx2.cpp)
else()
    message(FATAL_ERROR "Unsupported native platform: ${CMAKE_SYSTEM_NAME}/${CMAKE_SYSTEM_PROCESSOR}")
endif()

add_library(accel_core STATIC src/dispatch.cpp src/transform.cpp src/scalar.cpp ${OMNIMORPH_KERNELS})
target_include_directories(accel_core PUBLIC include)
target_compile_definitions(accel_core PUBLIC ${OMNIMORPH_ARCH})

add_library(omnimorph_accel SHARED src/jni.cpp)
target_link_libraries(omnimorph_accel PRIVATE accel_core)
target_include_directories(omnimorph_accel PRIVATE ${JNI_INCLUDE_DIRS})
if(OMNIMORPH_JNI_HEADERS)
    target_include_directories(omnimorph_accel PRIVATE "${OMNIMORPH_JNI_HEADERS}")
    target_compile_definitions(omnimorph_accel PRIVATE OMNIMORPH_GENERATED_JNI)
endif()
set_target_properties(omnimorph_accel PROPERTIES
    RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin"
    LIBRARY_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")

add_executable(accel_tests tests/accel_tests.cpp)
target_link_libraries(accel_tests PRIVATE accel_core)
add_test(NAME acceleration_native COMMAND accel_tests)

foreach(target accel_core omnimorph_accel accel_tests)
    if(MSVC)
        target_compile_options(${target} PRIVATE /W4 /fp:strict)
    else()
        target_compile_options(${target} PRIVATE -Wall -Wextra -Wpedantic -fno-fast-math -ffp-contract=off)
    endif()
endforeach()

if(OMNIMORPH_ARCH STREQUAL "OMNIMORPH_X64")
    # Keep loader/CPUID/scalar/SSE2 at baseline ISA; do not require FMA in any kernel.
    if(MSVC)
        set_source_files_properties(src/avx2.cpp PROPERTIES COMPILE_OPTIONS "/arch:AVX2")
    else()
        target_compile_options(accel_core PRIVATE -march=x86-64 -mno-avx -mno-fma)
        set_source_files_properties(src/avx2.cpp PROPERTIES COMPILE_OPTIONS "-mavx2;-mno-fma")
    endif()
endif()
